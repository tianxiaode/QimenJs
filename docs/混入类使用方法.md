// EventMixin：注入事件能力
function EventMixin<T extends Constructor>(Base: T) {
  return class extends Base {
    private _eventBase = new EventBase();

    on = this._eventBase.on.bind(this._eventBase);
    off = this._eventBase.off.bind(this._eventBase);
    emit = this._eventBase.emit.bind(this._eventBase);
  };
}

// LoggerMixin：注入日志能力
function LoggerMixin<T extends Constructor>(Base: T) {
  return class extends Base {
    log(message: string) {
      console.log(`[LOG]: ${message}`);
    }
  };
}

// =========================
// EventBase 示例
// =========================
class EventBase {
  private listeners: Record<string, Function[]> = {};

  on(event: string, handler: Function) {
    if (!this.listeners[event]) this.listeners[event] = [];
    this.listeners[event].push(handler);
  }

  off(event: string, handler?: Function) {
    if (!handler) {
      delete this.listeners[event];
    } else {
      this.listeners[event] = (this.listeners[event] || []).filter(h => h !== handler);
    }
  }

  emit(event: string, ...args: any[]) {
    (this.listeners[event] || []).forEach(fn => fn(...args));
  }
}

// =========================
// 使用示例
// =========================
class BaseService {
  start() {
    console.log("Service started");
  }
}

// 组合多个混入类
const CombinedService = composeMixins(BaseService, [EventMixin, LoggerMixin]);

const service = new CombinedService();

// EventMixin 功能
service.on("done", () => console.log("Event done"));
service.emit("done"); // 输出: Event done

// LoggerMixin 功能
service.log("Hello world"); // 输出: [LOG]: Hello world

// BaseService 原有功能
service.start(); // 输出: Service started
